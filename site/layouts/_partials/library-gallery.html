{{/* Renders a GLightbox gallery from library entry frontmatter media. Expects .Params.media.cover and optional .Params.media.gallery (paths under library/). */}} {{- $media := .Params.media -}} {{- if $media -}} {{- $paths := slice -}} {{- with $media.cover -}} {{- $paths = $paths | append . -}} {{- end -}} {{- range $media.gallery -}} {{- $paths = $paths | append . -}} {{- end -}} {{- if gt (len $paths) 0 -}} {{- $galleryID := printf "lib-gal-%s" (.RelPermalink | anchorize) -}}
{{- range $i, $path := $paths -}} {{- $base := path.Base $path -}} {{- $dir := path.Dir $path -}} {{- $thumbName := printf "thumb-%s" $base -}} {{- $fullURL := printf "library/%s" $path | relURL -}} {{- $thumbURL := printf "library/%s/%s" $dir $thumbName | relURL -}} {{- end -}}
{{- end -}} {{- end -}}
